\section{Problem Description}
A single machine must process three production jobs. Each job, once started,
cannot be interrupted, and the machine handles at most one job at a time.
We seek the processing order that minimizes the total weighted completion time.

\begin{center}
\begin{tabular}{lcc}
\hline
Job $j$ & Processing time $p_j$ (h) & Weight $w_j$ \\
\hline
1 & 2 & 3 \\
2 & 3 & 1 \\
3 & 1 & 2 \\
\hline
\end{tabular}
\end{center}

\section{Mathematical Model}

\subsection{Sets}
\[
\mathcal{J} = \{1, 2, 3\} \quad \text{(jobs)}, \qquad
\mathcal{K} = \{1, 2, 3\} \quad \text{(positions)}
\]

\subsection{Parameters}
\[
p_j \in \mathbb{Z}_+ : \text{processing time of job } j \text{ (hours)}, \qquad
w_j \in \mathbb{Z}_+ : \text{weight (priority) of job } j
\]

\subsection{Decision Variables}
\[
x_{jk} \in \{0, 1\}, \quad \forall j \in \mathcal{J},\ k \in \mathcal{K}
\]
where $x_{jk} = 1$ iff job $j$ is scheduled at position $k$.

\subsection{Objective Function}
\begin{equation}
\min \sum_{k \in \mathcal{K}} C_k \sum_{j \in \mathcal{J}} w_j \, x_{jk}
\label{eq:objective}
\end{equation}

\subsection{Constraints}
\begin{align}
\sum_{k \in \mathcal{K}} x_{jk} &= 1, \quad \forall j \in \mathcal{J}
  \label{eq:one-position-per-job} \\
\sum_{j \in \mathcal{J}} x_{jk} &= 1, \quad \forall k \in \mathcal{K}
  \label{eq:one-job-per-position} \\
C_k &= \sum_{l=1}^{k} \sum_{j \in \mathcal{J}} p_j \, x_{jl}, \quad \forall k \in \mathcal{K}
  \label{eq:completion-time}
\end{align}

\subsection{Remarks}
\begin{itemize}
  \item \textbf{Complexity}: $\mathcal{P}$ (polynomially solvable).
        Optimal order is given by WSPT / Smith's rule: sort jobs by
        $p_j / w_j$ ascending. The exact solver is used for
        cross-validation.
  \item \textbf{Numerical hygiene}: no big-$M$ constants; all parameters are
        integer hours (no scaling needed); CP-SAT solves to proven optimality.
  \item \textbf{Solver}: OR-Tools CP-SAT 9.x (project default stack).
\end{itemize}
